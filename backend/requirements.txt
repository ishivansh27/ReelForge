yt-dlp==2026.7.4

# Video analysis -- scene cut detection (CPU-only, no GPU needed)

librosa==0.11.0
# av (PyAV) bundles its own FFmpeg libraries inside the wheel, so we can
# decode audio out of the downloaded .mp4 without a separate system
# ffmpeg install.
av==18.0.0

# AI asset matching (CLIP embeddings + face detection).
# torch is installed from the CPU-only wheel index -- avoids pulling
# down CUDA libraries we can't use anyway (no dedicated GPU on this
# laptop). Runs fine on CPU for single-image embeddings; this is much
# lighter than Whisper/Demucs, which is why it wasn't deferred like
# those were. Install with:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#   pip install open_clip_torch
torch==2.13.0
open_clip_torch==3.3.0
# scipy comes in as a librosa dependency already; used here for
# scipy.optimize.linear_sum_assignment (optimal slot<->asset matching).

# Text overlay OCR -- requires the Tesseract-OCR system binary too
# (installed separately, e.g. `winget install UB-Mannheim.TesseractOCR`
# on Windows, `apt install tesseract-ocr` on Linux). pytesseract is
# just a thin wrapper around that binary, not an OCR engine itself.
pytesseract==0.3.13

# Payments (used later, harmless to install now)
stripe==11.4.1

# ---- Dev/test only ----
# moto simulates real AWS services locally so we can test the S3 upload

# account. Not used by the running app itself.
moto[s3]==5.0.28
